import jwt from 'jsonwebtoken';
import { redis } from '../lib/redis';
import { UnauthorizedError } from '../errors';

export async function verifyRefreshToken(token: string) {
  const payload = jwt.verify(token, process.env.REFRESH_SECRET!) as { sub: string };
  const userKey = `refresh:${payload.sub}`;
  const storedToken = await redis.get(userKey);

  if (!storedToken || storedToken !== token) {
    await redis.del(userKey);
    throw new UnauthorizedError("Reused refresh token detected");
  }

  return payload;
}
